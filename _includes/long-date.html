{%- comment -%} Renders a date as "Sunday, 5th July, 2026". Usage: {% include long-date.html date=post.date %} {%- endcomment -%} {%- assign day = include.date | date: "%-d" -%} {%- case day -%} {%- when '1' or '21' or '31' -%}{%- assign suffix = 'st' -%} {%- when '2' or '22' -%}{%- assign suffix = 'nd' -%} {%- when '3' or '23' -%}{%- assign suffix = 'rd' -%} {%- else -%}{%- assign suffix = 'th' -%} {%- endcase -%} {{ include.date | date: '%A,' }} {{ day }}{{ suffix }} {{ include.date | date: '%B, %Y' -}}
